%% List spacing %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% book.cls leaves \flushbottom on for a twoside document and the default
% list glue is elastic, so a page that runs short is padded out by
% stretching the gaps between list items. Rigid spacing so a list cannot
% be stretched, and \raggedbottom so LaTeX stops trying to.
\setlist{topsep=4pt, itemsep=2pt, parsep=0pt, partopsep=0pt}
\raggedbottom

% Row height for every table in the document. It lives here rather than in the
% body so a chapter compiled on its own as a subfile gets the same tables as
% the full build.
\renewcommand{\arraystretch}{1.25}

% The prose is full of unbreakable code tokens, which leave TeX no legal break
% and a line in the margin. This buys a last pass with looser spacing instead.
\setlength{\emergencystretch}{3em}

tabsize=2,                      % sets default tabsize to 2 spaces
breaklines=true,                % sets automatic line breaking
breakatwhitespace=false,        % sets if automatic breaks should only happen at whitespace
}
